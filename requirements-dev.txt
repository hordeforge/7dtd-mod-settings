# Developer and CI tools, pinned, installed with:
#
#   python3 -m pip install -r requirements-dev.txt
#
# scripts/lint-python.sh runs both, and .github/workflows/ci.yml installs
# this file, so a local green run and a remote green run mean the same thing.
# This is the only place a version is written down;
# scripts/test_lint_toolchain_declared.py fails if a pin is repeated in
# README, the workflow or any other tracked file, where it would drift.
#
# The closure is pinned too, not just the two tools. `pip install mypy`
# resolves five more distributions on its own, and an unpinned one of those
# is a release this repository never reviewed entering the CI lint lane
# (and every contributor's machine) on the day it is published. The
# versions below are the ones the toolchain is verified against; the two
# mypy needs only below 3.11 carry the marker that scopes them.
ruff==0.16.4
mypy==2.3.1
mypy_extensions==1.1.0
typing_extensions==4.16.0
pathspec==1.1.1
librt==0.15.0
